#!/usr/bin/env python3

import argparse
import datetime
import json
import os
import re
import socketserver
import threading
import time
import uuid
from pathlib import Path

import serial


class DailyLogger:
    def __init__(self, directory):
        self.dir = Path(directory).expanduser()
        self.dir.mkdir(parents=True, exist_ok=True)
        self.lock = threading.Lock()
        self.day = None
        self.logf = None
        self.jsonf = None

    def _open(self):
        day = datetime.datetime.now().astimezone().strftime("%Y-%m-%d")
        if day == self.day and self.logf and self.jsonf:
            return

        if self.logf:
            self.logf.close()
        if self.jsonf:
            self.jsonf.close()

        self.day = day
        log_path = self.dir / f"{day}.log"
        json_path = self.dir / f"{day}.jsonl"

        self.logf = open(log_path, "a", encoding="utf-8", buffering=1)
        self.jsonf = open(json_path, "a", encoding="utf-8", buffering=1)

        latest = self.dir / "latest.log"
        try:
            latest.unlink()
        except FileNotFoundError:
            pass
        latest.symlink_to(log_path.name)

    def write(self, kind, data="", actor=None, event=None):
        with self.lock:
            self._open()

            now = datetime.datetime.now().astimezone()
            human_ts = now.strftime("%Y-%m-%d %H:%M:%S.%f")[:-3]
            iso_ts = now.isoformat(timespec="milliseconds")

            if kind == "rx":
                tag = "RX"
            elif kind == "tx":
                tag = f"TX:{actor or 'unknown'}"
            else:
                tag = "EVENT"

            if event:
                text = event
                if data:
                    text += f" {data}"
            else:
                text = data

            # human-readable log
            parts = text.replace("\r", "").split("\n")
            if not parts:
                parts = [""]

            for part in parts:
                if part == "" and len(parts) > 1:
                    continue
                self.logf.write(f"[{human_ts}] [{tag}] {part}\n")

            # machine-readable log
            obj = {
                "ts": iso_ts,
                "type": kind,
            }

            if actor:
                obj["actor"] = actor
            if event:
                obj["event"] = event
            if data:
                obj["data"] = data

            self.jsonf.write(
                json.dumps(obj, ensure_ascii=False) + "\n"
            )


class SerialState:
    MAX_HISTORY = 2 * 1024 * 1024

    def __init__(self, device, baud, logger):
        self.device = device
        self.baud = baud
        self.logger = logger

        self.ser = None
        self.connected = False
        self.resolved_device = None

        self.serial_lock = threading.Lock()
        self.history_lock = threading.Condition()
        self.history = ""
        self.base_seq = 0
        self.end_seq = 0

    def current_seq(self):
        with self.history_lock:
            return self.end_seq

    def append_rx(self, text):
        with self.history_lock:
            self.history += text
            self.end_seq += len(text)

            if len(self.history) > self.MAX_HISTORY:
                remove = len(self.history) - self.MAX_HISTORY
                self.history = self.history[remove:]
                self.base_seq += remove

            self.history_lock.notify_all()

    def text_since(self, seq):
        with self.history_lock:
            if seq < self.base_seq:
                seq = self.base_seq

            offset = seq - self.base_seq
            return self.history[offset:]

    def wait_for(self, pattern, start_seq, timeout, regex=False):
        deadline = time.monotonic() + timeout

        with self.history_lock:
            while True:
                if start_seq < self.base_seq:
                    start_seq = self.base_seq

                offset = start_seq - self.base_seq
                text = self.history[offset:]

                if regex:
                    match = re.search(pattern, text, re.MULTILINE)
                    if match:
                        return True, text
                else:
                    if pattern in text:
                        return True, text

                left = deadline - time.monotonic()
                if left <= 0:
                    return False, text

                self.history_lock.wait(timeout=min(left, 0.5))

    def send(self, text, actor="unknown"):
        data = text.encode("utf-8")

        with self.serial_lock:
            if not self.ser or not self.connected:
                raise RuntimeError("serial device is not connected")

            self.ser.write(data)
            self.ser.flush()

        self.logger.write("tx", text, actor=actor)

    def status(self):
        return {
            "connected": self.connected,
            "device": self.device,
            "resolved_device": self.resolved_device,
            "baud": self.baud,
            "rx_seq": self.current_seq(),
        }


STATE = None


def serial_worker():
    global STATE

    last_wait_log = 0

    while True:
        if STATE.ser is None:
            try:
                resolved = os.path.realpath(STATE.device)

                ser = serial.Serial(
                    port=STATE.device,
                    baudrate=STATE.baud,
                    bytesize=serial.EIGHTBITS,
                    parity=serial.PARITY_NONE,
                    stopbits=serial.STOPBITS_ONE,
                    timeout=0.2,
                    write_timeout=2,
                )

                STATE.ser = ser
                STATE.connected = True
                STATE.resolved_device = resolved

                STATE.logger.write(
                    "event",
                    data=f"{STATE.device} -> {resolved} baud={STATE.baud}",
                    event="CONNECT",
                )

            except Exception as e:
                now = time.monotonic()
                if now - last_wait_log > 30:
                    STATE.logger.write(
                        "event",
                        data=str(e),
                        event="WAITING_DEVICE",
                    )
                    last_wait_log = now

                time.sleep(1)
                continue

        try:
            data = STATE.ser.read(4096)

            if data:
                text = data.decode("utf-8", errors="replace")
                STATE.append_rx(text)
                STATE.logger.write("rx", text)

        except Exception as e:
            STATE.logger.write(
                "event",
                data=str(e),
                event="DISCONNECT",
            )

            with STATE.serial_lock:
                try:
                    if STATE.ser:
                        STATE.ser.close()
                except Exception:
                    pass

                STATE.ser = None
                STATE.connected = False
                STATE.resolved_device = None

            time.sleep(1)


class Handler(socketserver.StreamRequestHandler):
    def send_response(self, obj):
        self.wfile.write(
            (json.dumps(obj, ensure_ascii=False) + "\n").encode("utf-8")
        )
        self.wfile.flush()

    def handle(self):
        line = self.rfile.readline()
        if not line:
            return

        try:
            req = json.loads(line.decode("utf-8"))
            cmd = req.get("cmd")

            if cmd == "status":
                self.send_response({
                    "ok": True,
                    **STATE.status(),
                })
                return

            if cmd == "send":
                text = req.get("text", "")
                actor = req.get("actor", "unknown")
                STATE.send(text, actor)
                self.send_response({"ok": True})
                return

            if cmd == "wait":
                pattern = req["pattern"]
                timeout = float(req.get("timeout", 30))
                regex = bool(req.get("regex", False))

                start_seq = req.get("start_seq")
                if start_seq is None:
                    start_seq = STATE.current_seq()

                ok, text = STATE.wait_for(
                    pattern,
                    int(start_seq),
                    timeout,
                    regex,
                )

                self.send_response({
                    "ok": ok,
                    "matched": ok,
                    "data": text,
                })
                return

            if cmd == "exec":
                command = req["command"]
                actor = req.get("actor", "unknown")
                timeout = float(req.get("timeout", 30))

                token = uuid.uuid4().hex
                marker = f"__SC_{token}__"

                start_seq = STATE.current_seq()

                # marker is deliberately assembled by printf, so terminal echo
                # does not contain the complete marker and cause a false hit.
                payload = (
                    f"{command}\n"
                    f"__sc_rc=$?\n"
                    f"printf '\\n__SC_%s__:%s\\n' '{token}' \"$__sc_rc\"\n"
                )

                STATE.send(payload, actor)

                ok, text = STATE.wait_for(
                    marker + ":",
                    start_seq,
                    timeout,
                    False,
                )

                if not ok:
                    self.send_response({
                        "ok": False,
                        "error": "timeout waiting for command marker",
                        "data": text,
                    })
                    return

                m = re.search(
                    re.escape(marker) + r":(-?\d+)",
                    text,
                )

                rc = int(m.group(1)) if m else None

                output = text
                if m:
                    output = text[:m.start()]

                self.send_response({
                    "ok": True,
                    "rc": rc,
                    "data": output,
                })
                return

            self.send_response({
                "ok": False,
                "error": f"unknown command: {cmd}",
            })

        except Exception as e:
            self.send_response({
                "ok": False,
                "error": str(e),
            })


class ThreadingUnixServer(socketserver.ThreadingUnixStreamServer):
    daemon_threads = True


def main():
    global STATE

    ap = argparse.ArgumentParser()
    ap.add_argument("--device", required=True)
    ap.add_argument("--baud", type=int, required=True)
    ap.add_argument("--log-dir", required=True)
    ap.add_argument("--socket", required=True)
    args = ap.parse_args()

    logger = DailyLogger(args.log_dir)
    STATE = SerialState(args.device, args.baud, logger)

    socket_path = os.path.expanduser(args.socket)
    os.makedirs(os.path.dirname(socket_path), exist_ok=True)

    try:
        os.unlink(socket_path)
    except FileNotFoundError:
        pass

    logger.write(
        "event",
        data=f"device={args.device} baud={args.baud}",
        event="SERIALD_START",
    )

    threading.Thread(
        target=serial_worker,
        daemon=True,
    ).start()

    with ThreadingUnixServer(socket_path, Handler) as server:
        os.chmod(socket_path, 0o600)

        try:
            server.serve_forever()
        finally:
            try:
                os.unlink(socket_path)
            except FileNotFoundError:
                pass


if __name__ == "__main__":
    main()

#!/usr/bin/env python3

import argparse
import collections
import json
import os
import re
import socket
import subprocess
import sys
from pathlib import Path


DEFAULT_SOCKET = os.path.join(
    os.environ.get("XDG_RUNTIME_DIR", f"/run/user/{os.getuid()}"),
    "seriald-rv1126b.sock",
)

DEFAULT_LOG_DIR = Path.home() / "work/serial-logs/rv1126b"


def request(obj, socket_path):
    s = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)

    try:
        s.connect(socket_path)
    except FileNotFoundError:
        print(
            f"seriald socket 不存在: {socket_path}\n"
            f"检查: systemctl --user status seriald-rv1126b",
            file=sys.stderr,
        )
        sys.exit(2)
    except ConnectionRefusedError:
        print("seriald 没有运行", file=sys.stderr)
        sys.exit(2)

    with s:
        s.sendall(
            (json.dumps(obj, ensure_ascii=False) + "\n").encode("utf-8")
        )

        data = b""
        while not data.endswith(b"\n"):
            chunk = s.recv(65536)
            if not chunk:
                break
            data += chunk

    if not data:
        raise RuntimeError("seriald returned no response")

    return json.loads(data.decode("utf-8"))


def actor(args):
    return (
        args.actor
        or os.environ.get("SERIAL_ACTOR")
        or "human"
    )


def latest_log():
    return DEFAULT_LOG_DIR / "latest.log"


def main():
    ap = argparse.ArgumentParser(
        prog="serialctl",
        description="RV1126B serial control client",
    )

    ap.add_argument(
        "--socket",
        default=DEFAULT_SOCKET,
    )

    ap.add_argument(
        "--actor",
        help="TX source name, e.g. human/luna/codex/opencode",
    )

    sp = ap.add_subparsers(dest="command", required=True)

    sp.add_parser("status")

    p = sp.add_parser("send")
    p.add_argument("text")
    p.add_argument(
        "--raw",
        action="store_true",
        help="do not append newline",
    )

    p = sp.add_parser("wait")
    p.add_argument("pattern")
    p.add_argument("-t", "--timeout", type=float, default=30)
    p.add_argument("--regex", action="store_true")

    p = sp.add_parser("exec")
    p.add_argument("text")
    p.add_argument("-t", "--timeout", type=float, default=30)

    p = sp.add_parser("tail")
    p.add_argument("lines", nargs="?", type=int, default=100)

    sp.add_parser("watch")

    p = sp.add_parser("grep")
    p.add_argument("pattern")
    p.add_argument("-i", "--ignore-case", action="store_true")

    args = ap.parse_args()

    if args.command == "status":
        r = request({"cmd": "status"}, args.socket)

        print(f"connected : {r.get('connected')}")
        print(f"device    : {r.get('device')}")
        print(f"resolved  : {r.get('resolved_device')}")
        print(f"baud      : {r.get('baud')}")
        print(f"socket    : {args.socket}")
        print(f"log       : {latest_log()}")
        return

    if args.command == "send":
        text = args.text
        if not args.raw:
            text += "\n"

        r = request({
            "cmd": "send",
            "text": text,
            "actor": actor(args),
        }, args.socket)

        if not r.get("ok"):
            print(r.get("error"), file=sys.stderr)
            sys.exit(1)
        return

    if args.command == "wait":
        r = request({
            "cmd": "wait",
            "pattern": args.pattern,
            "timeout": args.timeout,
            "regex": args.regex,
        }, args.socket)

        if r.get("data"):
            print(r["data"], end="")

        if not r.get("matched"):
            print(
                f"\n等待超时: {args.pattern}",
                file=sys.stderr,
            )
            sys.exit(1)
        return

    if args.command == "exec":
        r = request({
            "cmd": "exec",
            "command": args.text,
            "timeout": args.timeout,
            "actor": actor(args),
        }, args.socket)

        if r.get("data"):
            print(r["data"], end="")

        if not r.get("ok"):
            print(
                "\n" + r.get("error", "exec failed"),
                file=sys.stderr,
            )
            sys.exit(1)

        rc = r.get("rc")
        if rc is not None and rc != 0:
            sys.exit(rc)
        return

    if args.command == "tail":
        path = latest_log()

        if not path.exists():
            print(f"日志不存在: {path}", file=sys.stderr)
            sys.exit(1)

        q = collections.deque(maxlen=args.lines)

        with open(path, "r", encoding="utf-8", errors="replace") as f:
            for line in f:
                q.append(line)

        print("".join(q), end="")
        return

    if args.command == "watch":
        path = latest_log()

        if not path.exists():
            print(f"日志不存在: {path}", file=sys.stderr)
            sys.exit(1)

        os.execvp(
            "tail",
            ["tail", "-F", str(path)],
        )

    if args.command == "grep":
        path = latest_log()

        flags = re.IGNORECASE if args.ignore_case else 0
        rx = re.compile(args.pattern, flags)

        if not path.exists():
            print(f"日志不存在: {path}", file=sys.stderr)
            sys.exit(1)

        with open(path, "r", encoding="utf-8", errors="replace") as f:
            for line in f:
                if rx.search(line):
                    print(line, end="")
        return


if __name__ == "__main__":
    main()

### 跑完以后怎么用

重新打开终端后，先：

```bash
serialctl status
```

理想状态应该是：

```text
connected : True
device    : /dev/serial/by-id/usb-FTDI_FT232R_USB_UART_AL0783ER-if00-port0
resolved  : /dev/ttyUSB0
baud      : 1500000
...
```

然后你可以自己开一个窗口实时看：

```bash
serialctl watch
```

另外一个窗口发：

```bash
serialctl send 'uname -a'
```

看最近 100 行：

```bash
serialctl tail 100
```

查错误：

```bash
serialctl grep -i 'error|fail|panic'
```

等待**之后新出现的**登录提示：

```bash
serialctl wait 'login:' -t 60
```

如果开发板当前已经登录进 shell，最有用的是：

```bash
serialctl exec 'uname -a'
```

或者：

```bash
serialctl exec 'cat /etc/os-release'
```

`exec` 会自动在命令后面塞一个唯一结束标记，所以它知道“这条命令什么时候执行完”，不需要猜 shell 提示符。

### Luna / Codex / OpenCode 怎么调用

它们不用任何 MCP，直接执行 CLI 就行。

为了让日志里能看出是谁发送的：

```bash
serialctl --actor luna exec 'uname -a'

serialctl --actor codex exec 'dmesg | tail -n 30'

serialctl --actor opencode send 'reboot'
```

日志就会记录成：

```text
[2026-09-07 19:32:10.123] [TX:luna] uname -a
[2026-09-07 19:32:10.126] [RX] Linux elf1126b-debian ...
```

也可以让 Agent 会话启动时设：

```bash
export SERIAL_ACTOR=luna
```

之后它只需要：

```bash
serialctl exec 'uname -a'
```

就会自动记成 `TX:luna`。

有一个重要限制：**以后不要再同时用 `tio / minicom / screen / SecureCRT` 直接打开 `/dev/ttyUSB0`。** 真串口应该只让 `seriald` 占用；你自己想看就 `serialctl watch`，Agent 也统一走 `serialctl`。这样日志、收发和多客户端才不会互相抢串口。

不要为了更新工具而盲目重新执行前面的整块 `cat > seriald` / `cat > serialctl` 脚本；它会覆盖当前版本的实时交互功能。修改代码后需要让 systemd 重新加载 `seriald` 时执行：

```bash
systemctl --user restart seriald-rv1126b.service
```

### 13.1 默认实时交互

```bash
serialctl attach
```

`attach` 默认是底部固定输入行模式：

- 串口异步输出显示在上方滚动区。
- 输入行固定在终端最底部，不会被串口数据冲散。
- 输入内容在本地编辑，按 Enter 后发送整行。
- 支持左右移动、Home/End、Backspace、Delete 和上下历史。
- 使用 `Ctrl+]` 退出。
- 识别到板端 `user@host:/path#`、`%` 或 `$` 提示符后，底部同步用户、主机、当前路径和提示符字符；`#` 为加粗红色，`%` 为较平缓的加粗蓝色，命令首词为加粗青色。
- 板端提示符和已发送命令回显也会高亮；少量高频网络状态使用快速高亮：`[network.c][rk_network_get_cable_state]` 为 256 色 `214`，`[wlan0] link down` 为加粗橙色；错误关键词为红色，成功/启动节点为绿色或青色，高频 `MD: md_area` 信息为暗灰色。
- 高亮只改变固定模式的终端显示，不改变串口日志文件、JSONL 或 `--raw` 输出。

### 13.2 原始逐字节模式

需要每个按键立即发送时使用：

```bash
serialctl attach --raw
```

原始模式适合 U-Boot、密码提示、全屏交互程序以及依赖逐字节输入的程序。原始模式不会提供底部固定输入行。

### 13.3 实时交互实现要点

- `seriald` 仍然是唯一直接打开真实串口的进程。
- `attach` 通过 Unix socket 与 `seriald` 建立长连接。
- TX 和 RX 以 Base64 传输，避免非 UTF-8 串口字节被破坏。
- `serialctl attach` 和 `serialctl attach --raw` 共用同一个串口会话，不要再启动 `screen`、`minicom`、`tio` 或 SecureCRT 抢占设备。
- `serialctl exec` 仍用于已进入 Linux shell 的自动化命令；不要用固定输入模式代替 `exec`。
